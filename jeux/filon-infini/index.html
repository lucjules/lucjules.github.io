Comment le coder sur Roblox

Ce jeu est un bon modèle de simulator Roblox. La règle d’or : le serveur décide de tout. Le client affiche et demande, il ne calcule jamais un achat ni un reset. Voici l’architecture en 4 scripts.

  • ModuleScript de config dans ServerStorage : les paliers et tous les nœuds (Id, Cost, Req, Effect). Les clients ne peuvent pas le lire.
  • Script serveur : profil de chaque joueur, achat vérifié via un RemoteEvent, découverte des nœuds.
  • Prestige : gain en racine carrée, reset, et logique « garder » en chaîne.
  • LocalScript : crée une carte uniquement pour les nœuds que le serveur lui annonce.
1 · Configuration (ModuleScript)

2 · Achat vérifié côté serveur (RemoteEvent)
-- ServerScriptService/GameServer  (Script)
-- Le serveur est la seule source de vérité : le client DEMANDE, le serveur DÉCIDE.
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerStorage = game:GetService("ServerStorage")

local Config = require(ServerStorage.GameConfig)

local Remotes = ReplicatedStorage:WaitForChild("Remotes")
local BuyNode = Remotes:WaitForChild("BuyNode")               -- RemoteEvent client -> serveur
local NodeDiscovered = Remotes:WaitForChild("NodeDiscovered") -- RemoteEvent serveur -> client

local Profiles = {} -- [Player] = profil (à sauvegarder avec DataStore / ProfileService)

local function newProfile()
	return {
		Currency = { [0] = 0, [1] = 0, [2] = 0, [3] = 0, [4] = 0 },
		Earned = { [0] = 0, [1] = 0, [2] = 0, [3] = 0, [4] = 0 }, -- gagné pendant CE run
		Owned = {},      -- [nodeId] = true
		Discovered = {}, -- [nodeId] = true, reste vrai même après un reset
		TierSeen = {},   -- [tier] = true
	}
end

local function reqsOwned(profile, node)
	for _, reqId in ipairs(node.Req) do
		if not profile.Owned[reqId] then
			return false
		end
	end
	return true
end

-- Un palier apparaît quand on atteint 50 % de son coût, la première fois
local function tierSeen(profile, tier)
	if profile.TierSeen[tier] then
		return true
	end
	local cfg = Config.Tiers[tier]
	if profile.Earned[tier - 1] >= cfg.Threshold * cfg.RevealAt then
		profile.TierSeen[tier] = true
		return true
	end
	return false
end

-- Révèle les nœuds dont TOUS les prérequis sont possédés.
-- On n'envoie au client que ces nœuds-la, jamais le reste.
local function refreshDiscovery(player, profile)
	for _, node in ipairs(Config.Nodes) do
		if not profile.Discovered[node.Id] and tierSeen(profile, node.Tier) and reqsOwned(profile, node) then
			profile.Discovered[node.Id] = true
			NodeDiscovered:FireClient(player, {
				Id = node.Id, Tier = node.Tier, Name = node.Name,
				Cost = node.Cost, Req = node.Req,
			})
		end
	end
end

-- Achat : toutes les vérifications sont faites ici, jamais côté client
BuyNode.OnServerEvent:Connect(function(player, nodeId)
	if typeof(nodeId) ~= "string" then return end        -- 1. type de l'argument
	local profile = Profiles[player]
	local node = Config.NodeById[nodeId]
	if not profile or not node then return end            -- 2. le nœud existe
	if profile.Owned[nodeId] then return end              -- 3. pas déjà acheté
	if not profile.Discovered[nodeId] then return end     -- 4. visible pour ce joueur
	if not reqsOwned(profile, node) then return end       -- 5. prérequis possédés
	if profile.Currency[node.Tier] < node.Cost then return end -- 6. assez de monnaie

	profile.Currency[node.Tier] -= node.Cost
	profile.Owned[nodeId] = true
	refreshDiscovery(player, profile)
end)

Players.PlayerAdded:Connect(function(player)
	Profiles[player] = newProfile() -- ici : charger depuis le DataStore
	refreshDiscovery(player, Profiles[player])
	-- à la connexion, renvoyer aussi les nœuds déjà découverts (jamais les autres)
end)
Players.PlayerRemoving:Connect(function(player)
	Profiles[player] = nil -- ici : sauvegarder
end)
3 · Reset et « garder » en chaîne
-- Suite de GameServer : prestige, reset et "garder"
local DoPrestige = Remotes:WaitForChild("DoPrestige") -- RemoteEvent client -> serveur

-- Garder l'arbre `lowTier` quand on reset `resetTier` ne marche que si
-- TOUS les nœuds « garder » des paliers entre les deux sont possédés.
-- Exemple : garder l'arbre 1 lors d'une Transcendance (3) exige A_Keep ET T_Keep.
local function keepsTree(profile, lowTier, resetTier)
	for tier = lowTier + 1, resetTier do
		if not profile.Owned[Config.KeepNode[tier]] then
			return false
		end
	end
	return true
end

local function gainMultiplier(profile, tier)
	local mult = Config.Tiers[tier].BaseMult
	for _, node in ipairs(Config.Nodes) do
		local fx = node.Effect
		if profile.Owned[node.Id] and fx.Type == "GainMult" and fx.Tier == tier then
			mult *= fx.Value
		end
	end
	return mult
end

-- gain = floor( racine( gagné pendant ce run ÷ seuil ) x multiplicateurs )
local function pendingGain(profile, tier)
	local cfg = Config.Tiers[tier]
	local earned = profile.Earned[tier - 1] -- GAGNÉ pendant ce run, pas la monnaie possédée
	if earned < cfg.Threshold then
		return 0
	end
	return math.floor(math.sqrt(earned / cfg.Threshold) * gainMultiplier(profile, tier))
end

local function wipeTree(profile, tier)
	for _, node in ipairs(Config.Nodes) do
		if node.Tier == tier then
			profile.Owned[node.Id] = nil -- on efface Owned, PAS Discovered
		end
	end
	if tier == 0 then
		profile.CoinUpgrades = {}
		profile.Generators = {}
	end
end

DoPrestige.OnServerEvent:Connect(function(player, tier)
	if typeof(tier) ~= "number" or not Config.Tiers[tier] then return end
	local profile = Profiles[player]
	if not profile then return end

	local gain = pendingGain(profile, tier)
	if gain < 1 then return end

	-- 1. décider quoi garder AVANT d'effacer quoi que ce soit
	local keep = {}
	for lower = 0, tier - 1 do
		keep[lower] = keepsTree(profile, lower, tier)
	end

	-- 2. créditer la monnaie du palier
	profile.Currency[tier] += gain
	profile.Earned[tier] += gain

	-- 3. remettre à zéro tout ce qui est en dessous
	for lower = 0, tier - 1 do
		profile.Currency[lower] = 0
		profile.Earned[lower] = 0
		if not keep[lower] then
			wipeTree(profile, lower)
		end
	end

	refreshDiscovery(player, profile)
end)

-- Boucle serveur (RunService.Heartbeat) : automatisations et gain passif
--   passif : profile.Currency[t-1] += 0.10 * pendingRaw(t-1) * dt
--   auto-prestige : si la condition choisie par le joueur est vraie, appeler la même fonction que DoPrestige
4 · Cacher les nœuds non découverts côté client
-- StarterPlayerScripts/TreeUI  (LocalScript)
-- Le client ne contient AUCUNE donnée sur les nœuds cachés.
-- Il crée une carte uniquement quand le serveur lui envoie NodeDiscovered.
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local TweenService = game:GetService("TweenService")
local Players = game:GetService("Players")

local Remotes = ReplicatedStorage:WaitForChild("Remotes")
local gui = Players.LocalPlayer:WaitForChild("PlayerGui"):WaitForChild("TreeGui")
local treeFrame = gui:WaitForChild("Tree")
local template = script:WaitForChild("NodeTemplate")

local cards = {} -- [nodeId] = Frame, uniquement les nœuds découverts

local function drawLine(fromCard, toCard)
	-- un Frame fin, tourné entre les deux cartes (ou une UIStroke / Beam)
end

local function createCard(info)
	if cards[info.Id] then return end -- une fois découvert, il reste affiché

	local card = template:Clone()
	card.Name = info.Id
	card.NodeName.Text = info.Name
	card.Cost.Text = tostring(info.Cost)
	card.Parent = treeFrame
	cards[info.Id] = card

	-- petite animation d'apparition
	local target = card.Size
	card.Size = UDim2.fromScale(0, 0)
	TweenService:Create(card, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), { Size = target }):Play()

	-- lignes : seulement vers les prérequis déjà découverts
	for _, reqId in ipairs(info.Req) do
		if cards[reqId] then
			drawLine(cards[reqId], card)
		end
	end

	-- notification
	gui.Toast:Fire("Nouvelle amélioration découverte !")
end

Remotes.NodeDiscovered.OnClientEvent:Connect(createCard)

-- Achat : le bouton n'achète rien lui-même, il demande au serveur
local function onNodeClicked(nodeId)
	Remotes.BuyNode:FireServer(nodeId)
end

Points importants

  • Ne jamais envoyer un nœud caché au client. Un tricheur lit tout ce qui est dans ReplicatedStorage. On n’envoie un nœud que lorsque ses prérequis sont possédés, et on le garde dans Discovered pour qu’il reste visible après un reset.
  • Le gain se calcule sur la monnaie gagnée pendant le run (Earned), pas sur celle que le joueur possède, pour ne pas le punir quand il dépense.
  • Décide quoi garder avant d’effacer. Les nœuds « garder » vivent dans les arbres que l’on est en train de réinitialiser ; il faut lire leur état avant le reset.
  • Production et automatisations tournent sur le serveur dans RunService.Heartbeat, avec le temps écoulé réel (dt) pour que la progression hors-ligne reste juste.
  • Sauvegarde avec DataStoreService (ou ProfileService) à la déconnexion et toutes les 60 secondes. Garde la monnaie en number ; au-delà de 1e300, passe à une bibliothèque de grands nombres.
Sauvegarde automatique dans ce navigateur. Les boutons ×10 et ×100 accélèrent le temps pour tester.
